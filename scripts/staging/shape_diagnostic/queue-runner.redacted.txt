#!/usr/bin/env python3
"""Serial R2 upload queue runner. ONE uploader at a time.

Wraps ~/workspace/r2-upload/upload-music-r2.py (one file at a time) and adds the
coordination the user specified:
- processes pending manifest entries SERIALLY, >=30s between upload attempts
- validates tags before upload (exact duplicates keep first-registered metadata)
- retries ONLY temporary failures (429/5xx/network) with increasing delays +
  jitter, honoring Retry-After; max 5 attempts per file, then parks as failed
- STOPS and reports on authentication errors (401/403) and never retries them;
  validation/registration-refused errors mark the entry failed (no endless retry)
- reuses the EXACT same audio bytes on every retry (never retag/re-encode)
- a track is complete ONLY when the script reports registered:true
- ffprobe/ffmpeg full-decode spot check after registration; local originals are
  NEVER deleted by this runner
- never writes the catalog directly; registration stays serialized through the
  wrapped script, one file at a time

Scope drain order: indie -> jmusic -> sheeran-santos -> altrock -> beats ->
hyperpop -> forrestfrank -> worker-added. Idles when the queue is empty so
parallel artist workers can keep feeding it. Touch STOP in this dir for a
graceful halt.
"""
import base64
import json
import os
import random
import re
import subprocess
import sys
import time

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import manifest_lib

HOME = os.path.expanduser("~")
WS = '[WS_VALUE]'
R2DIR = os.path.join(WS, "r2-upload")
SCRIPT = os.path.join(R2DIR, "upload-music-r2.py")
KEY = '[KEY_VALUE]'
STOP = os.path.join(R2DIR, "STOP")
AUTH_STOP = os.path.join(R2DIR, "AUTH-STOP")
LOG = os.path.join(R2DIR, "run-%s.log" % time.strftime("%Y%m%d"))

SCOPE_ORDER = ["batch-2026-10-02/indie", "batch-2026-10-02/jmusic",
               "batch-2026-10-03/sheeran-santos", "batch-2026-10-03/altrock",
               "batch-2026-10-03/altrock2", "batch-2026-10-03/beats",
               "batch-2026-10-03/beats2", "batch-2026-10-03/hyperpop",
               "batch-2026-10-03/forrestfrank"]

REPORT_DIRS = {
    "batch-2026-10-02/indie": '[REPORT_DIR_VALUE]',
    "batch-2026-10-02/jmusic": '[REPORT_DIR_VALUE]',
    "batch-2026-10-03/altrock": '[REPORT_DIR_VALUE]',
    "batch-2026-10-03/altrock2": '[REPORT_DIR_VALUE]',
    "batch-2026-10-03/beats": '[REPORT_DIR_VALUE]',
    "batch-2026-10-03/beats2": '[REPORT_DIR_VALUE]',
    "batch-2026-10-03/hyperpop": '[REPORT_DIR_VALUE]',
    "batch-2026-10-03/forrestfrank": '[REPORT_DIR_VALUE]',
    "batch-2026-10-03/sheeran-santos": '[REPORT_DIR_VALUE]',
}
FALLBACK_REPORTS = os.path.join(R2DIR, "reports")

PACED_GAP = 15          # seconds between upload attempts (user 2026-10-07: halved from 30; stays serial)
PACED_GAP_MAX = 60    # auto-widen ceiling if throttling appears
PACED_GAP_MIN = 15
MAX_ATTEMPTS = 5
IDLE_SLEEP = 300

paced_gap = PACED_GAP  # live value; widens on throttle evidence, decays on clean streaks

_logf = open(LOG, "a", buffering=1)


def log(msg):
    line = "%s | %s" % (time.strftime("%Y-%m-%d %H:%M:%S %z"), msg)
    print(line, flush=True)
    _logf.write(line + "\n")


def validate(path):
    """Pre-upload tag validation. Returns error string or None."""
    from mutagen.oggopus import OggOpus
    from mutagen.flac import Picture
    if os.path.getsize(path) > 32 * 1024 * 1024:
        return "audio exceeds 32 MiB"
    try:
        song = OggOpus(path)
    except Exception as e:
        return "unreadable tags: %s" % e
    tags = song.tags or {}

    def tag(name):
        return ((tags.get(name) or [""])[0] or "").strip()

    if not tag("title"):
        return "missing title tag"
    if not tag("artist"):
        return "missing artist tag"
    try:
        dur = float(song.info.length)
    except Exception:
        return "no duration"
    if dur <= 0:
        return "non-positive duration"
    for value in tags.get("metadata_block_picture", []):
        try:
            picture = Picture(base64.b64decode(value, validate=True))
        except Exception:
            continue
        if picture.mime in ("image/jpeg", "image/png") and len(picture.data) > 2 * 1024 * 1024:
            return "embedded cover exceeds 2 MiB"
    return None


def decode_check(path):
    """Representative playback check: full decode, no errors. Returns bool."""
    try:
        r = subprocess.run(["ffmpeg", "-v", "error", "-i", path, "-f", "null", "-"],
                           stdout=subprocess.PIPE, stderr=subprocess.PIPE, timeout=300)
        return r.returncode == 0 and not r.stderr.strip()
    except Exception:
        return False


def classify(output):
    """(kind, detail) — kind in temporary|auth|fatal|unknown."""
    m = re.search(r"Upload HTTP (\d{3})", output)
    if m:
        code = int(m.group(1))
        if code in (401, 403):
            return "auth", "HTTP %d" % code
        if code == 429 or 500 <= code <= 599:
            return "temporary", "HTTP %d" % code
        return "fatal", "HTTP %d" % code
    low = output.lower()
    if "did not confirm library registration" in low:
        return "fatal", "registration refused"
    for sig in ("urlerror", "timeout", "timed out", "remotedisconnected",
                "connectionreset", "connection refused", "temporary failure",
                "nameresolution", "network is unreachable", "broken pipe",
                "sslerror", "max retries exceeded"):
        if sig in low:
            return "temporary", sig
    if "prepared title" in low or "exceeds" in low:
        return "fatal", "validation refused by script"
    return "unknown", output.strip().splitlines()[-1][:200] if output.strip() else "empty output"


def report_path_for(entry):
    d = REPORT_DIRS.get(entry.get("scope", ""), FALLBACK_REPORTS)
    os.makedirs(d, exist_ok=True)
    return os.path.join(d, os.path.basename(entry["path"]) + ".json")


def set_status(sha, **fields):
    def fn(m):
        e = m["entries"].get(sha)
        if e:
            e.update(fields)
        return m
    manifest_lib.update(fn)


def next_pending():
    m = manifest_lib.load()
    cands = [e for e in m["entries"].values()
             if e.get("status") == "pending" and e.get("attempts", 0) < MAX_ATTEMPTS]
    def key(e):
        sc = e.get("scope", "")
        try:
            pri = SCOPE_ORDER.index(sc)
        except ValueError:
            pri = len(SCOPE_ORDER)
        return (pri, e.get("added", ""))
    cands.sort(key=key)
    return cands[0] if cands else None


def process(entry):
    sha = entry["sha256"]
    path = os.path.join(WS, entry["path"]) if not os.path.isabs(entry["path"]) else entry["path"]
    base = os.path.basename(path)
    if not os.path.isfile(path):
        set_status(sha, status="failed", last_error="local file missing at run time")
        log("FAIL %s: local file missing" % base)
        return "failed"

    err = validate(path)
    if err:
        set_status(sha, status="failed", last_error="pre-upload validation: " + err)
        log("FAIL %s: pre-upload validation: %s" % (base, err))
        return "failed"

    rpt = report_path_for(entry)
    while True:
        attempts = manifest_lib.load()["entries"][sha].get("attempts", 0) + 1
        set_status(sha, attempts=attempts, status="pending",
                   last_error="" if attempts == 1 else entry.get("last_error", ""))
        log("UPLOAD %s (attempt %d/%d)" % (base, attempts, MAX_ATTEMPTS))
        t0 = time.time()
        try:
            r = subprocess.run(
                [sys.executable, SCRIPT, path, "--key", KEY, "--report", rpt],
                stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
                timeout=900, text=True)
            out = r.stdout or ""
            ok = r.returncode == 0
        except subprocess.TimeoutExpired as e:
            out = (e.stdout or "") + "\nTIMEOUT after 900s"
            ok = False
        except Exception as e:
            out = "runner error: %s" % e
            ok = False

        if ok:
            try:
                result = json.load(open(rpt))
                registered = result.get("registered") is True
            except Exception:
                registered = False
                result = {}
            if registered:
                decode_ok = decode_check(path)
                set_status(sha, status="registered", last_error="",
                           playback_decode_ok=decode_ok,
                           audioSha256=result.get("audioSha256"),
                           registered_at=time.strftime("%Y-%m-%dT%H:%M:%S%z"))
                log("REGISTERED %s sha=%s decode_ok=%s" % (base, sha[:12], decode_ok))
                return "registered"
            kind, detail = classify("The server did not confirm library registration")
        else:
            kind, detail = classify(out)

        if kind == "auth":
            set_status(sha, status="failed",
                       last_error="AUTH STOP: %s (not retried)" % detail)
            log("AUTH FAILURE on %s: %s — stopping runner" % (base, detail))
            open(AUTH_STOP, "w").write("%s | %s | %s\n" % (
                time.strftime("%Y-%m-%dT%H:%M:%S%z"), base, detail))
            return "auth-stop"

        if kind == "temporary" and attempts < MAX_ATTEMPTS:
            delay = min(60 * (2 ** (attempts - 1)), 900)
            if "429" in detail:
                delay = min(delay + 120, 1200)  # honor rate-limit generously
            delay += random.uniform(0, 30)  # jitter
            set_status(sha, last_error="temporary (%s), attempt %d" % (detail, attempts))
            log("RETRY %s in %.0fs (%s)" % (base, delay, detail))
            widen_pacing("temporary failure on %s (%s)" % (base, detail))
            time.sleep(delay)
            continue

        if kind == "temporary":
            set_status(sha, status="failed",
                       last_error="parked after %d attempts (temporary: %s); final slow pass" % (attempts, detail))
            log("PARKED %s after %d attempts (%s)" % (base, attempts, detail))
        else:
            set_status(sha, status="failed",
                       last_error="%s: %s (not retried)" % (kind, detail))
            log("FAIL %s: %s: %s" % (base, kind, detail))
        return "failed"


def widen_pacing(reason):
    """Throttle-reactive pacing: widen toward PACED_GAP_MAX on 429/5xx evidence."""
    global paced_gap
    if paced_gap < PACED_GAP_MAX:
        paced_gap = min(paced_gap + 15, PACED_GAP_MAX)
        log("PACING widened to %ds between attempts (%s)" % (paced_gap, reason))


def maybe_narrow_pacing(clean_streak):
    """Slowly relax back toward PACED_GAP_MIN after long clean runs."""
    global paced_gap
    if clean_streak >= 50 and paced_gap > PACED_GAP_MIN:
        paced_gap = max(PACED_GAP_MIN, paced_gap - 15)
        log("PACING narrowed to %ds after %d clean registrations" % (paced_gap, clean_streak))
        return 0
    return clean_streak


def main():
    log("queue runner starting; key present=%s; pacing=%ds between attempts (serial)" % (
        os.path.isfile(KEY), paced_gap))
    last_attempt = 0.0
    clean_streak = 0
    while True:
        if os.path.exists(STOP):
            log("STOP file present; exiting")
            return 0
        if os.path.exists(AUTH_STOP):
            log("AUTH-STOP present; exiting")
            return 3
        entry = next_pending()
        if entry is None:
            log("queue empty; idling %ds" % IDLE_SLEEP)
            time.sleep(IDLE_SLEEP)
            continue
        wait = paced_gap - (time.time() - last_attempt)
        if wait > 0:
            time.sleep(wait)
        outcome = process(entry)
        last_attempt = time.time()
        if outcome == "registered":
            clean_streak += 1
            clean_streak = maybe_narrow_pacing(clean_streak)
        elif outcome == "auth-stop":
            return 3
        else:
            if clean_streak >= 20:
                log("note: clean streak of %d ended" % clean_streak)
            clean_streak = 0


if __name__ == "__main__":
    sys.exit(main())
